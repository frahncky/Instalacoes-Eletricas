  \draw[white,line width=5.2pt] (#1,#3)--(#2,#3);
  \draw[black!20,thin] (#1,#3)--(#2,#3);}
\newcommand{\ArqVaoV}[3]{%
  \draw[white,line width=5.2pt] (#3,#1)--(#3,#2);

  \draw[white,line width=5.4pt] (#1,#3)--(#2,#3);
